    function showEmpty() {
      result.className = 'result empty';
      result.replaceChildren();
      const icon = document.createElement('div');
      icon.className = 'placeholder-icon';
      icon.textContent = 'Aa';
      const title = document.createElement('div');

    function searchWord(raw) {
      const query = raw.trim().toLowerCase();
      if (!query) {
        showEmpty();
        input.focus();
        return;
      }
      const entry = byWord.get(query);
      result.replaceChildren();
      if (!entry) {
        result.className = 'result';
        const label = document.createElement('div');
        label.className = 'result-label not-found';
        label.textContent = 'NOT FOUND';
        const title = document.createElement('div');
        title.className = 'empty-title';
        title.style.marginTop = '9px';
        title.textContent = 'この単語は見つかりませんでした';
        const copy = document.createElement('div');
        copy.className = 'empty-copy';
        copy.textContent = '綴りを確認するか、PDFに載っている見出し語を入力してください。';
        result.append(label, title, copy);
        return;
      }

      result.className = 'result';
      const label = document.createElement('div');
      label.className = 'result-label';
      label.textContent = 'WORD FOUND';
      const line = document.createElement('div');
      line.className = 'word-line';
      const word = document.createElement('div');
      word.className = 'word';
      word.textContent = entry.word;
      const number = document.createElement('div');
      number.className = 'number';
      number.textContent = `NO. ${entry.number}`;
      line.append(word, number);
      const meaning = document.createElement('div');
      meaning.className = 'meaning';
      meaning.textContent = entry.meaning;
      result.append(label, line, meaning);
    }
